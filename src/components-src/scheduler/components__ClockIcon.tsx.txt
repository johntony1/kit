import { useEffect, useRef } from 'react'
import gsap from 'gsap'

/* ─────────────────────────────────────────────────────────
 * ANIMATION STORYBOARD — ClockIcon
 *
 *  mount     outer ring: scale 0.8 → 1 with spring (GSAP elastic)
 *  mount     outer ring: opacity 0 → 1 (160ms)
 *  ∞ loop    minute hand: real-time sweep rotation (360° / 60s)
 *  ∞ loop    hour hand:   real-time sweep rotation (360° / 43200s)
 *  ∞ loop    outer body:  gentle scale pulse 1 ↔ 1.03 (4s yoyo)
 *  hover     whole clock: scale 1.1, drop shadow brightens
 * ───────────────────────────────────────────────────────── */

function getClockAngles() {
  const now = new Date()
  const h = now.getHours() % 12
  const m = now.getMinutes()
  const s = now.getSeconds()
  const minuteDeg = m * 6 + s * 0.1         // 360° / 60min
  const hourDeg   = h * 30 + m * 0.5        // 360° / 12h
  return { minuteDeg, hourDeg, s }
}

export default function ClockIcon() {
  const outerRef  = useRef<SVGGElement>(null)
  const minuteRef = useRef<SVGGElement>(null)
  const hourRef   = useRef<SVGGElement>(null)
  const rootRef   = useRef<SVGSVGElement>(null)

  useEffect(() => {
    const { minuteDeg, hourDeg, s } = getClockAngles()
    const mRem = 60 - s                      // seconds remaining in current minute

    if (!minuteRef.current || !hourRef.current || !outerRef.current) return

    // — entrance spring
    gsap.fromTo(rootRef.current,
      { scale: 0.72, opacity: 0 },
      { scale: 1, opacity: 1, duration: 0.8, ease: 'elastic.out(1, 0.55)', delay: 0.05 }
    )

    // — set initial hand angles (svgOrigin = clock center)
    gsap.set(minuteRef.current, { svgOrigin: '20 20', rotation: minuteDeg })
    gsap.set(hourRef.current,   { svgOrigin: '20 20', rotation: hourDeg })

    // — continuous real-time sweeps
    gsap.to(minuteRef.current, {
      rotation: minuteDeg + 360,
      duration: 60,
      ease: 'none',
      repeat: -1,
      delay: mRem - 60,  // sync to real minute
    })
    gsap.to(hourRef.current, {
      rotation: hourDeg + 360,
      duration: 43200,
      ease: 'none',
      repeat: -1,
      delay: -( (Date.now() / 1000) % 43200 ),
    })

    // — outer body pulse
    gsap.to(outerRef.current, {
      scale: 1.03,
      duration: 3.5,
      ease: 'sine.inOut',
      repeat: -1,
      yoyo: true,
      svgOrigin: '20 20',
    })

    return () => gsap.killTweensOf([rootRef.current, minuteRef.current, hourRef.current, outerRef.current])
  }, [])

  return (
    <svg
      ref={rootRef}
      width="40"
      height="40"
      viewBox="0 0 40 40"
      fill="none"
      xmlns="http://www.w3.org/2000/svg"
      style={{ display: 'block', transformOrigin: '20px 20px' }}
    >
      <defs>
        <filter id="sh-outer" x="-10%" y="-10%" width="120%" height="120%">
          <feDropShadow dx="0" dy="0.16" stdDeviation="0.08" floodOpacity="0.08"/>
          <feDropShadow dx="0" dy="0"    stdDeviation="0.08" floodOpacity="0.28"/>
        </filter>
        <filter id="sh-face" x="-8%" y="-8%" width="116%" height="116%">
          <feDropShadow dx="0" dy="0" stdDeviation="0.73" floodOpacity="0.12"/>
        </filter>
        <filter id="sh-hand" x="-80%" y="-10%" width="260%" height="120%">
          <feDropShadow dx="0" dy="0" stdDeviation="0.59" floodOpacity="0.08"/>
        </filter>
        <filter id="sh-center" x="-60%" y="-60%" width="220%" height="220%">
          <feDropShadow dx="0" dy="0" stdDeviation="0.74" floodOpacity="0.12"/>
        </filter>
        <linearGradient id="grad-body" x1="52.19" y1="13.45" x2="13.55" y2="-12.45" gradientUnits="userSpaceOnUse">
          <stop stopColor="#3374F6"/>
          <stop offset="1" stopColor="#00D3ED"/>
        </linearGradient>
        <linearGradient id="grad-stroke" x1="18.71" y1="2.47" x2="21.15" y2="37.39" gradientUnits="userSpaceOnUse">
          <stop stopColor="#F8F8F8"/>
          <stop offset="1" stopColor="#D0D0D0"/>
        </linearGradient>
        <clipPath id="body-clip">
          <path fillRule="evenodd" clipRule="evenodd" d="M19.128 2.587C27.852 0.892 34.61 5.45 36.306 14.174L37.269 19.128C38.964 27.852 34.406 34.61 25.682 36.306L20.728 37.269C12.005 38.964 5.246 34.406 3.55 25.682L2.588 20.728C0.892 12.005 5.45 5.246 14.174 3.55Z"/>
        </clipPath>
      </defs>

      {/* — decorative outer body */}
      <g ref={outerRef} filter="url(#sh-outer)">
        <path
          d="M19.219 3.056C23.478 2.228 27.215 2.933 30.086 4.869C32.956 6.806 35.01 10.007 35.838 14.265L36.801 19.219C37.628 23.478 36.923 27.215 34.987 30.086C33.051 32.956 29.849 35.01 25.591 35.837L20.637 36.801C16.379 37.628 12.641 36.923 9.77 34.987C6.9 33.05 4.847 29.849 4.019 25.591L3.056 20.637C2.228 16.379 2.933 12.641 4.869 9.77C6.806 6.9 10.007 4.846 14.265 4.019Z"
          fill="url(#grad-body)"
          stroke="url(#grad-stroke)"
          strokeWidth="0.954"
        />
        {/* dot texture overlay */}
        <g clipPath="url(#body-clip)">
          <path d="M7.084 7.807C4.535 9.949 3.211 13.581 3.211 15.517C4.61 15.23 28.636 10.353 33.621 9.277C31.541 5.548 27.991 3.934 25.624 3.396C23.257 2.858 20.783 3.145 16.91 3.934C12.877 4.755 10.455 4.974 7.084 7.807Z" fill="black" fillOpacity="0.12"/>
          {[
            [7.55,18.69],[11.49,17.93],[15.42,17.16],[19.37,16.39],[23.31,15.63],
            [8.42,23.20],[12.36,22.43],[16.30,21.67],[20.24,20.90],[24.19,20.13],[28.13,19.37],[32.07,18.60],
            [9.30,27.70],[13.24,26.94],[17.17,26.17],[21.12,25.40],[25.06,24.64],[29.01,23.87],[32.95,23.11],
            [10.15,32.21],[14.10,31.45],[18.06,30.68],[22.00,29.91],[25.94,29.14],[29.88,28.38],[33.82,27.61],
          ].map(([cx, cy], i) => (
            <circle key={i} cx={cx} cy={cy} r="1.148" transform={`rotate(-11 ${cx} ${cy})`} fill="black" fillOpacity="0.12"/>
          ))}
        </g>
      </g>

      {/* — clock face */}
      <g filter="url(#sh-face)">
        <ellipse cx="19.93" cy="19.94" rx="12.60" ry="12.80" transform="rotate(3 19.93 19.94)" fill="white"/>
      </g>

      {/* — minute hand (long, thin) */}
      <g ref={minuteRef} filter="url(#sh-hand)">
        <rect
          x="19.531"
          y="8.5"
          width="0.938"
          height="11.8"
          rx="0.469"
          fill="#222222"
        />
      </g>

      {/* — hour hand (short, thick) */}
      <g ref={hourRef} filter="url(#sh-hand)">
        <rect
          x="19.2"
          y="12"
          width="1.6"
          height="8.2"
          rx="0.804"
          fill="#222222"
        />
      </g>

      {/* — center jewel */}
      <g filter="url(#sh-center)">
        <circle cx="19.917" cy="20.011" r="1.34" transform="rotate(3 19.917 20.011)" fill="white"/>
      </g>
    </svg>
  )
}
