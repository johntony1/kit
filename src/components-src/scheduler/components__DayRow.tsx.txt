import { useState } from 'react'
import { motion, AnimatePresence } from 'framer-motion'
import { PencilFill, CheckLine } from '@mingcute/react'
import Toggle from './Toggle'
import TimePicker, { type TimeValue, HOURS, MINUTES } from './TimePicker'

/* ─────────────────────────────────────────────────────────
 * STORYBOARD — DayRow interactions
 *
 * Toggle ON  → panel springs open (height 0→auto)
 * Pencil     → display→edit: pills split from center (x±24)
 * Time pill  → picker springs open below (scaleY 0→1, origin top)
 * Other pill → picker cross-fades to new time context
 * Check      → picker closes, edit→display: pills merge to center
 * Toggle OFF → everything collapses
 * ─────────────────────────────────────────────────────────*/

const EXPAND_SPRING = { type: 'spring', stiffness: 380, damping: 30 } as const
const ITEM_SPRING   = { type: 'spring', stiffness: 420, damping: 26 } as const
const FAST_SPRING   = { type: 'spring', stiffness: 520, damping: 24 } as const
const BOUNCE_SPRING = { type: 'spring', stiffness: 460, damping: 20 } as const

interface DayRowProps {
  label: string
  checked: boolean
  onChange: (v: boolean) => void
  index: number
}

export const rowVariants = {
  hidden: { opacity: 0, y: 10 },
  visible: (i: number) => ({
    opacity: 1, y: 0,
    transition: { delay: 0.32 + i * 0.075, duration: 0.28, ease: [0.25, 0.46, 0.45, 0.94] as [number, number, number, number] },
  }),
}

const DEFAULT_START: TimeValue = { hour: 7,  minute: 0, period: 'AM' }  // 08:00 AM (index 7 → "08")
const DEFAULT_END:   TimeValue = { hour: 9,  minute: 0, period: 'PM' }  // 10:00 PM (index 9 → "10")

const PILL_STYLE = {
  backgroundColor: '#f7f7f7',
  borderRadius: 12,
  padding: '8px 12px',
  display: 'flex',
  alignItems: 'center',
  justifyContent: 'space-between',
  minWidth: 0,
  border: 'none',
  cursor: 'pointer',
} as const

function formatTime(v: TimeValue) {
  return `${HOURS[v.hour]}:${MINUTES[v.minute]}`
}

export default function DayRow({ label, checked, onChange, index }: DayRowProps) {
  const [editing, setEditing]       = useState(false)
  const [activeTime, setActiveTime] = useState<'start' | 'end' | null>(null)
  const [start, setStart]           = useState<TimeValue>(DEFAULT_START)
  const [end, setEnd]               = useState<TimeValue>(DEFAULT_END)

  const handleToggle = (val: boolean) => {
    if (!val) { setEditing(false); setActiveTime(null) }
    onChange(val)
  }

  const handleCheck = () => {
    setEditing(false)
    setActiveTime(null)
  }

  const handlePillClick = (slot: 'start' | 'end') => {
    setActiveTime(prev => prev === slot ? null : slot)
  }

  const activeValue = activeTime === 'start' ? start : end
  const setActive   = activeTime === 'start' ? setStart : setEnd

  return (
    <motion.div
      custom={index}
      variants={rowVariants}
      initial="hidden"
      animate="visible"
      layout
      transition={{ layout: EXPAND_SPRING }}
      style={{ borderRadius: 12 }}
    >
      <motion.div
        layout
        animate={{
          backgroundColor: checked ? 'rgba(255,255,255,0)' : '#f7f7f7',
          boxShadow: checked ? 'inset 0 0 0 1px #ebebeb' : 'inset 0 0 0 1px rgba(235,235,235,0)',
          paddingTop: 8, paddingBottom: 8,
          paddingLeft: checked ? 8 : 0,
          paddingRight: checked ? 8 : 0,
        }}
        transition={EXPAND_SPRING}
        style={{ borderRadius: 12 }}
      >
        {/* toggle header */}
        <motion.div
          layout
          animate={{ paddingLeft: checked ? 4 : 12, paddingRight: checked ? 4 : 12 }}
          transition={EXPAND_SPRING}
          style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}
        >
          <span style={{ fontSize: 14, fontWeight: 500, letterSpacing: '-0.084px', lineHeight: '20px', whiteSpace: 'nowrap', userSelect: 'none', color: '#171717' }}>
            {label}
          </span>
          <Toggle checked={checked} onChange={handleToggle} />
        </motion.div>

        {/* expanded panel */}
        <AnimatePresence initial={false}>
          {checked && (
            <motion.div
              key="panel"
              initial={{ height: 0, opacity: 0 }}
              animate={{ height: 'auto', opacity: 1 }}
              exit={{ height: 0, opacity: 0 }}
              transition={EXPAND_SPRING}
              style={{ overflow: 'hidden' }}
            >
              <div style={{ height: 12 }} />

              {/* display ↔ edit swap */}
              <div style={{ position: 'relative' }}>
                <AnimatePresence mode="popLayout" initial={false}>
                  {!editing ? (
                    /* ── DISPLAY: single pill + pencil ── */
                    <motion.div
                      key="display"
                      initial={{ opacity: 0, scale: 0.88 }}
                      animate={{ opacity: 1, scale: 1, transition: { ...ITEM_SPRING, delay: 0.06 } }}
                      exit={{ opacity: 0, scaleX: 0.55, transition: FAST_SPRING }}
                      style={{ ...PILL_STYLE, cursor: 'default', justifyContent: 'space-between' }}
                    >
                      <span style={{ fontSize: 14, fontWeight: 500, letterSpacing: '-0.084px', lineHeight: '20px', color: '#171717', fontVariantNumeric: 'tabular-nums' }}>
                        {formatTime(start)} {start.period} – {formatTime(end)} {end.period}
                      </span>
                      <motion.button
                        onClick={() => setEditing(true)}
                        aria-label="Edit time"
                        initial={{ scale: 0, rotate: -30 }}
                        animate={{ scale: 1, rotate: 0, transition: { ...BOUNCE_SPRING, delay: 0.1 } }}
                        exit={{ scale: 0, rotate: 90, transition: FAST_SPRING }}
                        whileHover={{ scale: 1.15 }}
                        whileTap={{ scale: 0.88 }}
                        style={{ background: 'none', border: 'none', cursor: 'pointer', padding: 2, display: 'flex', alignItems: 'center', justifyContent: 'center', flexShrink: 0 }}
                      >
                        <PencilFill size={16} color="#a3a3a3" />
                      </motion.button>
                    </motion.div>
                  ) : (
                    /* ── EDIT: two time pickers + check ── */
                    <motion.div key="edit" style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
                      {/* pills row */}
                      <div style={{ display: 'flex', gap: 8, alignItems: 'stretch' }}>
                        {/* start time pill */}
                        <motion.button
                          onClick={() => handlePillClick('start')}
                          initial={{ x: 24, opacity: 0, scale: 0.88 }}
                          animate={{ x: 0, opacity: 1, scale: 1, boxShadow: activeTime === 'start' ? 'inset 0 0 0 1.5px #3592f9' : 'inset 0 0 0 0px #3592f9' }}
                          exit={{ x: 18, opacity: 0, scale: 0.9, transition: FAST_SPRING }}
                          transition={ITEM_SPRING}
                          style={{
                            ...PILL_STYLE,
                            flex: 1,
                            justifyContent: 'space-between',
                            gap: 6,
                          }}
                        >
                          <span style={{ fontSize: 14, fontWeight: 500, letterSpacing: '-0.084px', lineHeight: '20px', color: '#171717', fontVariantNumeric: 'tabular-nums' }}>
                            {formatTime(start)}
                          </span>
                          <span style={{ fontSize: 14, fontWeight: 500, letterSpacing: '-0.084px', lineHeight: '20px', color: '#a3a3a3' }}>
                            {start.period}
                          </span>
                        </motion.button>

                        {/* end time pill */}
                        <motion.button
                          onClick={() => handlePillClick('end')}
                          initial={{ x: -24, opacity: 0, scale: 0.88 }}
                          animate={{ x: 0, opacity: 1, scale: 1, boxShadow: activeTime === 'end' ? 'inset 0 0 0 1.5px #3592f9' : 'inset 0 0 0 0px #3592f9' }}
                          exit={{ x: -18, opacity: 0, scale: 0.9, transition: FAST_SPRING }}
                          transition={{ ...ITEM_SPRING, delay: 0.03 }}
                          style={{
                            ...PILL_STYLE,
                            flex: 1,
                            justifyContent: 'space-between',
                            gap: 6,
                          }}
                        >
                          <span style={{ fontSize: 14, fontWeight: 500, letterSpacing: '-0.084px', lineHeight: '20px', color: '#171717', fontVariantNumeric: 'tabular-nums' }}>
                            {formatTime(end)}
                          </span>
                          <span style={{ fontSize: 14, fontWeight: 500, letterSpacing: '-0.084px', lineHeight: '20px', color: '#a3a3a3' }}>
                            {end.period}
                          </span>
                        </motion.button>

                        {/* check button */}
                        <motion.button
                          onClick={handleCheck}
                          aria-label="Confirm"
                          initial={{ scale: 0, y: 3 }}
                          animate={{ scale: 1, y: 0 }}
                          exit={{ scale: 0, transition: FAST_SPRING }}
                          transition={{ ...BOUNCE_SPRING, delay: 0.09 }}
                          whileHover={{ scale: 1.08, backgroundColor: '#efefef' }}
                          whileTap={{ scale: 0.92 }}
                          style={{ ...PILL_STYLE, flexShrink: 0, padding: '8px 12px', cursor: 'pointer', transition: 'background-color 0.15s ease' }}
                        >
                          <CheckLine size={20} color="#a3a3a3" />
                        </motion.button>
                      </div>

                      {/* time picker drum — springs open when a pill is active */}
                      <AnimatePresence initial={false}>
                        {activeTime !== null && (
                          <motion.div
                            key="picker"
                            initial={{ scaleY: 0, opacity: 0 }}
                            animate={{ scaleY: 1, opacity: 1 }}
                            exit={{ scaleY: 0, opacity: 0 }}
                            transition={EXPAND_SPRING}
                            style={{ transformOrigin: 'top center', overflow: 'hidden' }}
                          >
                            <TimePicker
                              value={activeValue}
                              onChange={setActive}
                            />
                          </motion.div>
                        )}
                      </AnimatePresence>
                    </motion.div>
                  )}
                </AnimatePresence>
              </div>
            </motion.div>
          )}
        </AnimatePresence>
      </motion.div>
    </motion.div>
  )
}
