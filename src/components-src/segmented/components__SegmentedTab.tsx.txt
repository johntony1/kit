/* ─────────────────────────────────────────────────────────
 * ANIMATION STORYBOARD — Segmented Tab + Status Popover + Actions Panel
 *
 * Tab switch:
 *    0ms   white pill springs old → new tab (layoutId)
 *    0ms   button press scale 1 → 0.97, springs back on release
 *  150ms   labels cross-fade soft ↔ strong
 *  180ms   adjacent dividers fade out; far dividers fade in
 *
 * "Set status" opens popover:
 *    0ms   outer pill: scale 0.88→1, y +10→0, opacity 0→1 (spring)
 *   55ms   icon 1 bounces in  |  100ms icon 2  |  ...  |  235ms icon 5
 *  exit    all fade + scale 0.88 (120ms ease-in)
 *
 * Status icon selected → pill morph:
 *    0ms   white pill springs to selected pill size (layoutId morph)
 *   50ms   status icon bounces in (scale 0.3→1)
 *   80ms   label shimmer-slides in  |  110ms × springs in
 *  exit    pill springs back to "Set status" size
 *
 * "+" → container morphs into Actions panel (layout spring):
 *    0ms   tab bar fades out (80ms ease-in)
 *    0ms   container springs from tab-bar → actions-panel shape
 *   80ms   "Actions" header slides up (spring)
 *  120ms   row 1 springs up  |  160ms row 2  |  200ms row 3  |  240ms row 4
 *  exit    rows/header fade out (80ms), container springs back to tab-bar shape
 * ───────────────────────────────────────────────────────── */

type TabId      = "all" | "set-status" | "draft" | "archived";

function showDivider(left: TabId | null, right: TabId | null, active: TabId) {
  return active !== left && active !== right;

/* ─── ShimmerText ─── */
function ShimmerText({ children, doShimmer = true }: { children: React.ReactNode; doShimmer?: boolean }) {
  const reduced = useReducedMotion();
  const [on, setOn] = useState(doShimmer && !reduced);
  return <span className={on ? "text-shimmer" : undefined} onAnimationEnd={() => setOn(false)}>{children}</span>;
}

/* ─── StatusIconButton ─── */
function StatusIconButton({ src, label, getIsQuick, onHoverEnd, onSelect }: {
  src: string; label: string;
  getIsQuick: () => boolean; onHoverEnd: () => void; onSelect: () => void;
}) {
  const [hovered, setHovered] = useState(false);
  const [quick,   setQuick]   = useState(false);
  const reduced = useReducedMotion();

  return (
    <motion.div variants={POPOVER_ICON} className="relative shrink-0">
      <AnimatePresence>
        {hovered && (
          <motion.div key="tip" className="absolute bottom-full left-1/2 mb-2 z-[70]" style={{ transformOrigin: "bottom center" }}
            initial={reduced || quick ? false : TOOLTIP.initial} animate={TOOLTIP.animate}
            exit={reduced ? {} : TOOLTIP.exit} transition={reduced || quick ? { duration: 0 } : TOOLTIP.spring}>
            <TooltipBubble label={label} />
          </motion.div>
        )}
      </AnimatePresence>
      <motion.button aria-label={label} onClick={onSelect}
        onHoverStart={() => { setQuick(getIsQuick()); setHovered(true); }}
        onHoverEnd={() => { setHovered(false); onHoverEnd(); }}
        whileHover={reduced ? {} : { scale: 1.16 }} whileTap={reduced ? {} : { scale: 0.88 }}
        transition={{ type: "spring", stiffness: 500, damping: 20 }}
        className="flex items-center justify-center rounded-[24px] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-black/20"
        style={{ background: "#f7f7f7", padding: 6, WebkitTapHighlightColor: "transparent" }}>
        <img src={src} alt="" draggable={false} className="block size-5 object-cover pointer-events-none select-none" />
      </motion.button>
    </motion.div>
  );
}

/* ─── StatusPopover ─── */
function StatusPopover({ onSelect }: { onSelect: (icon: StatusIcon) => void }) {
  const lastHoverEndAt = useRef(0);
  const getIsQuick = () => Date.now() - lastHoverEndAt.current < TIMING.quickHoverMs;
  const handleHoverEnd = () => { lastHoverEndAt.current = Date.now(); };

  return (
    <motion.div variants={POPOVER_CONTAINER} initial="initial" animate="animate" exit="exit"
      style={{ transformOrigin: "bottom center" }}
      className="absolute bottom-full left-1/2 mb-3 z-50 w-max">
      <div className="flex items-center rounded-[32px]" style={{ background: COLOR.white, padding: 4, gap: 8, border: `0.5px solid ${COLOR.border}`, boxShadow: "0px 16px 48px rgba(14,18,27,0.14), 0px 6px 16px rgba(14,18,27,0.08), 0px 2px 6px rgba(14,18,27,0.06)" }}>
        {STATUS_ICONS.map(({ src, label }, i) => (
          <StatusIconButton key={i} src={src} label={label} getIsQuick={getIsQuick} onHoverEnd={handleHoverEnd} onSelect={() => onSelect({ src, label })} />
        ))}
      </div>
    </motion.div>
  );
}

/* ─── TabBtn ─── */
function TabBtn({ tabId, active, onClick, padding, children }: {
  tabId: TabId; active: TabId; onClick: () => void; padding: string; children: React.ReactNode;
}) {
  const isActive = active === tabId;
  const reduced  = useReducedMotion();
  return (
    <motion.button role="tab" aria-selected={isActive} onClick={onClick}
      whileTap={reduced ? {} : { scale: PRESS.scale }} transition={PRESS.spring}
      className={`relative flex items-center justify-center gap-1 ${padding} rounded-2xl focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-1 focus-visible:ring-black/20`}
      style={{ WebkitTapHighlightColor: "transparent" }}>
      {isActive && (
        <motion.span layoutId="tab-indicator" className="absolute inset-0"
          style={{ background: COLOR.white, boxShadow: INDICATOR.shadow, zIndex: 0, borderRadius: 16 }}
          transition={reduced ? { duration: 0 } : INDICATOR.spring} />
      )}
      <motion.span animate={{ color: isActive ? COLOR.strong : COLOR.soft }}
        transition={{ duration: TIMING.label, ease: "easeInOut" }}
        className="relative z-10 flex items-center gap-1 text-sm font-medium leading-5 select-none whitespace-nowrap"
        style={{ letterSpacing: "-0.084px", fontFeatureSettings: "'ss11' 1, 'calt' 0, 'liga' 0" }}>
        {children}
      </motion.span>
    </motion.button>
  );
}

/* ─── TabsView — the normal tab bar (one state of the morphing container) ─── */
function TabsView({ active, selectedStatus, popoverOpen, showSelectedPill, statusClearCount,
  onTabClick, onStatusTabClick, onStatusSelect, onClearStatus }: {
  active: TabId; selectedStatus: StatusIcon | null; popoverOpen: boolean;
  showSelectedPill: boolean; statusClearCount: number;
  onTabClick: (id: TabId) => void; onStatusTabClick: () => void;
  onStatusSelect: (icon: StatusIcon) => void; onClearStatus: () => void;
}) {
  return (
    <motion.div
      initial={false}
      exit={{ opacity: 0, transition: { duration: 0.08, ease: "easeIn" } }}
      role="tablist" aria-label="Content filter"
      className="inline-flex items-center gap-1"
      style={{ padding: 4 }}>

      <TabBtn tabId="all" active={active} onClick={() => onTabClick("all")} padding="px-3 py-1">All</TabBtn>
      <Divider visible={showDivider("all", "set-status", active)} />

      {/* Set status */}
      <div className="relative shrink-0">
        {showSelectedPill ? (
          <SelectedStatusBtn status={selectedStatus!} onOpen={onStatusTabClick} onClear={onClearStatus} />
        ) : (
          <TabBtn tabId="set-status" active={active} onClick={onStatusTabClick} padding="px-2 py-1">
            <ShimmerText key={statusClearCount} doShimmer={statusClearCount > 0}>Set status</ShimmerText>
            <ChevronIcon active={active === "set-status" && popoverOpen} />
          </TabBtn>
        )}
        <AnimatePresence>
          {active === "set-status" && popoverOpen && <StatusPopover onSelect={onStatusSelect} />}
        </AnimatePresence>
      </div>

      <Divider visible={showDivider("set-status", "draft", active)} />
      <TabBtn tabId="draft" active={active} onClick={() => onTabClick("draft")} padding="px-3 py-1">Draft</TabBtn>
      <Divider visible={showDivider("draft", "archived", active)} />
      <TabBtn tabId="archived" active={active} onClick={() => onTabClick("archived")} padding="px-2 py-1">Archived</TabBtn>
      <Divider visible={showDivider("archived", null, active)} />
      <PlusBtn />
    </motion.div>
  );
}

/* ─── SegmentedTab — main export ─── */
export function SegmentedTab() {
  const [active,           setActive]           = useState<TabId>("all");
  const [selectedStatus,   setSelectedStatus]   = useState<StatusIcon | null>(null);
  const [popoverOpen,      setPopoverOpen]      = useState(false);
  const [actionsOpen,      setActionsOpen]      = useState(false);
  const [statusClearCount, setStatusClearCount] = useState(0);

  function handleCloseActions() { setActionsOpen(false); }

  function handleStatusTabClick() {
    if (active === "set-status") { setPopoverOpen(p => !p); }
    else { setActive("set-status"); setPopoverOpen(true); }
  }

  function handleOtherTabClick(id: TabId) { setActive(id); setPopoverOpen(false); }

  function handleStatusSelect(icon: StatusIcon) {
    setSelectedStatus(icon); setPopoverOpen(false); setActive("set-status");
  }

  function handleClearStatus() {
    setSelectedStatus(null); setPopoverOpen(false); setStatusClearCount(c => c + 1);
  }

  const showSelectedPill = active === "set-status" && selectedStatus !== null;
  const reduced = useReducedMotion();

  return (
    /*
      layout — Framer Motion springs the container between tab-bar size and
      actions-panel size whenever actionsOpen flips. No manual width/height needed.
    */
    <motion.div
      layout
      transition={{ layout: reduced ? { duration: 0 } : (actionsOpen ? SPRING_EXPAND : SPRING_LAYOUT) }}
      className={actionsOpen ? "overflow-hidden" : undefined}
      style={{
        borderRadius: actionsOpen ? 28 : 20,
        background:   actionsOpen ? "#f7f7f7" : COLOR.track,
      }}>
      <AnimatePresence mode="popLayout" initial={false}>
        {actionsOpen ? (
          <ActionsView key="actions" onClose={handleCloseActions} />
        ) : (
          <TabsView
            key="tabs"
            active={active}
            selectedStatus={selectedStatus}
            popoverOpen={popoverOpen}
            showSelectedPill={showSelectedPill}
            statusClearCount={statusClearCount}
            onTabClick={handleOtherTabClick}
            onStatusTabClick={handleStatusTabClick}
            onStatusSelect={handleStatusSelect}
            onClearStatus={handleClearStatus}
          />
        )}
      </AnimatePresence>
    </motion.div>
  );
}
