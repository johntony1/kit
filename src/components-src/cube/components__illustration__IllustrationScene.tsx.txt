import { Canvas, useFrame } from '@react-three/fiber'
import { useEffect, useMemo, useRef, useState } from 'react'
import type { Group, Mesh, MeshBasicMaterial } from 'three'
import { ILLUSTRATION, STAGE } from '../../lib/design'
import { TIMING } from '../../lib/motion'
import type { SceneInput } from '../../hooks/useSceneInput'
import { useViewport } from '../../hooks/useViewport'
import { useFaceStage } from '../../lib/faceStage'
import { useObjectTextures, type ObjectTexture } from './useObjectTextures'

/* ── Scene constants ──────────────────────────────────────────────────────
 * The camera is set up so that a plane at z = 0 maps 1:1 to design pixels.
 * Every object is then pushed to its own z and *pre-scaled* by (D - z) / D,
 * which cancels the perspective divide exactly — so at rest the render is
 * pixel-identical to the flat Figma composite, and the depth only becomes
 * visible once the scene rotates.
 * ───────────────────────────────────────────────────────────────────────── */

/** Head-room around the illustration so rotated layers aren't clipped. */
const PAD = 170

/**
 * The canvas box, rounded to whole pixels on purpose.
 *
 * r3f measures the container with offsetWidth/offsetHeight (see the Canvas
 * below), and those are integers. The camera has to agree with the number r3f
 * actually reads, not with the fractional design box, or the world-to-pixel
 * mapping drifts by a third of a pixel across the whole illustration — enough
 * to take the diff against the DOM render from 2.5/255 to 14.
 */
const CANVAS_W = Math.round(ILLUSTRATION.width + PAD * 2)
const CANVAS_H = Math.round(ILLUSTRATION.height + PAD * 2)

const FOV = 30
/** Camera distance that makes z = 0 render 1:1 with design pixels. */
const CAMERA_Z = CANVAS_H / 2 / Math.tan((FOV / 2) * (Math.PI / 180))

/** Furthest object sits at -DEPTH, nearest at +DEPTH. */
const DEPTH = 130

const MAX_YAW = 0.13 // radians at full cursor deflection
const MAX_PITCH = 0.085
/** Extra pitch as the hero scrolls away, in radians per viewport height. */
const SCROLL_PITCH = 0.12

const RISE = 46 // entrance travel, design px
const ENTER_DURATION = 0.72 // seconds

const easeOutCubic = (t: number) => 1 - Math.pow(1 - t, 3)
const damp = (current: number, target: number, lambda: number, dt: number) =>
  current + (target - current) * (1 - Math.exp(-lambda * dt))

type PlacedObject = ObjectTexture & {
  /** pre-compensated plane size and position */
  width: number
  height: number
  x: number
  y: number
  z: number
  /** perspective pre-scale, reused by the entrance offset */
  k: number
  order: number
  delay: number
}

function place(items: ObjectTexture[]): PlacedObject[] {
  // Stagger back-to-front so the scene builds itself away from the camera
  // first — depth order, not paint order, which is what the eye reads.
  const byDepth = [...items].sort((a, b) => a.object.depth - b.object.depth)
  const rank = new Map(byDepth.map((it, i) => [it.object.id, i]))

  return items.map((item) => {
    const { object } = item
    const z = (object.depth - 0.5) * 2 * DEPTH
    const k = (CAMERA_Z - z) / CAMERA_Z
    // Straight from box pixels to world units, so the fraction of a pixel the
    // rounded box adds around the illustration cannot go unaccounted for.
    const cx = PAD + object.x + object.w / 2 - CANVAS_W / 2
    const cy = CANVAS_H / 2 - (PAD + object.y + object.h / 2)

    return {
      ...item,
      width: object.w * k,
      height: object.h * k,
      x: cx * k,
      y: cy * k,
      z,
      k,
      order: object.order,
      delay: TIMING.illustration / 1000 + (rank.get(object.id) ?? 0) * (TIMING.illustrationStagger / 1000),
    }
  })
}

function Scene({
  input,
  still,
  supersample,
  onReady,
}: {
  input: React.RefObject<SceneInput>
  still: boolean
  supersample: number
  onReady: () => void
}) {
  const textures = useObjectTextures(supersample)
  const placed = useMemo(() => (textures ? place(textures) : []), [textures])

  useEffect(() => {
    if (textures) onReady()
  }, [textures, onReady])

  const group = useRef<Group>(null)
  const meshes = useRef<Mesh[]>([])
  const started = useRef<number | null>(null)

  useFrame((state, delta) => {
    const g = group.current
    if (!g) return
    const dt = Math.min(delta, 1 / 30)

    /* Entrance */
    if (started.current === null) started.current = state.clock.elapsedTime
    for (let i = 0; i < placed.length; i++) {
      const mesh = meshes.current[i]
      const item = placed[i]
      if (!mesh || !item) continue

      const p = still
        ? 1
        : Math.min(1, Math.max(0, (state.clock.elapsedTime - started.current - item.delay) / ENTER_DURATION))
      const eased = easeOutCubic(p)

      mesh.position.y = item.y - (1 - eased) * RISE * item.k
      const material = mesh.material as MeshBasicMaterial
      material.opacity = eased
    }

    /* Pointer + scroll driven rotation */
    if (!still) {
      const { px, py, scroll } = input.current
      g.rotation.y = damp(g.rotation.y, px * MAX_YAW, 3.4, dt)
      g.rotation.x = damp(g.rotation.x, py * MAX_PITCH + scroll * SCROLL_PITCH, 3.4, dt)
    }
  })

  if (!textures) return null

  return (
    <group ref={group}>
      {placed.map((item, i) => (
        <mesh
          key={item.object.id}
          ref={(node) => {
            if (node) meshes.current[i] = node
          }}
          position={[item.x, item.y, item.z]}
          renderOrder={item.order}
        >
          <planeGeometry args={[item.width, item.height]} />
          <meshBasicMaterial
            map={item.texture}
            transparent
            // Figma's own paint order is the correct occlusion for an
            // isometric scene, so depth buffering is off and renderOrder rules.
            depthTest={false}
            depthWrite={false}
            toneMapped={false}
          />
        </mesh>
      ))}
    </group>
  )
}

/**
 * How many texels to rasterise per design pixel.
 *
 * The stage scales the whole hero, so a design pixel is rarely a device pixel.
 * Tracking the real ratio (plus a little headroom) keeps the illustration as
 * crisp as the DOM render without wasting memory on textures nobody can see.
 * Quantised to quarter steps so a window drag doesn't re-rasterise 46 canvases
 * on every frame.
 */
function useSupersample() {
  const measure = () => {
    if (typeof window === 'undefined') return 2
    // `?ss=N` pins the rate — handy when diffing against the DOM render.
    const override = Number(new URLSearchParams(window.location.search).get('ss'))
    if (Number.isFinite(override) && override > 0) return override
    const stageScale = window.innerWidth / STAGE.width
    // 2× headroom over device density. Measured against the DOM render, this
    // is where the diff stops improving: the planes rotate and composite
    // through the stage transform, so oversampling past 1:1 still pays.
    const wanted = stageScale * (window.devicePixelRatio || 1) * 2
    return Math.min(3, Math.max(1.5, Math.round(wanted * 4) / 4))
  }

  const [supersample, setSupersample] = useState(measure)

  useEffect(() => {
    const onResize = () => setSupersample(measure())
    window.addEventListener('resize', onResize)
    return () => window.removeEventListener('resize', onResize)
  }, [])

  return supersample
}

export function IllustrationScene({
  input,
  still,
  onReady,
}: {
  input: React.RefObject<SceneInput>
  still: boolean
  onReady: () => void
}) {
  const supersample = useSupersample()
  const { scale } = useViewport()
  const wrapper = useRef<HTMLDivElement>(null)
  const [onScreen, setOnScreen] = useState(true)

  // Inside a cube deck the hero lives in a fixed stage, so it intersects the
  // viewport forever — including while the reader is two sections further down.
  // The deck says whether the face is actually being painted; the observer
  // covers the ordinary stacked case.
  const face = useFaceStage()
  const visible = onScreen && face.visible

  // The scene animates continuously while on screen — pointer damping never
  // fully settles, so `demand` would mean invalidating every frame anyway.
  // Off screen it drops to `demand` and stops costing anything.
  useEffect(() => {
    const el = wrapper.current
    if (!el) return
    const observer = new IntersectionObserver(([entry]) => setOnScreen(entry.isIntersecting), {
      rootMargin: '120px',
    })
    observer.observe(el)
    return () => observer.disconnect()
  }, [])

  return (
    <div
      ref={wrapper}
      className="absolute [&_canvas]:!h-full [&_canvas]:!w-full [&>div]:!h-full [&>div]:!w-full"
      style={{ left: -PAD, top: -PAD, width: CANVAS_W, height: CANVAS_H }}
      aria-hidden
    >
      <Canvas
        // `flat` disables tone mapping — anything else would shift the exported
        // colours away from the design.
        flat
        // Measure the *layout* box, not the painted one.
        //
        // r3f sizes the drawing buffer and the camera aspect from the container's
        // getBoundingClientRect, which reports the on-screen quad — already
        // multiplied by the scale-stage, and, once this section is a face of the
        // cube deck, foreshortened by the fold as well. Mid-turn that rect
        // collapses towards a line, three.js resizes to match, and nothing ever
        // puts it back: the layout box never changed, so no observer fires again
        // and the scene stays squashed into a vertical smear. offsetWidth /
        // offsetHeight ignore transforms entirely and are constant here.
        resize={{ offsetSize: true, scroll: false, debounce: 0 }}
        // Which hands density back to us: the buffer is now sized in design px,
        // so the stage factor has to be folded into dpr to keep one texel per
        // device pixel. (At 1512 wide the factor is 1 and this is the old [1, 2].)
        dpr={[1, Math.min(3, Math.max(1, (window.devicePixelRatio || 1) * scale))]}
        gl={{ antialias: true, alpha: true, premultipliedAlpha: false }}
        camera={{ fov: FOV, position: [0, 0, CAMERA_Z], near: 1, far: CAMERA_Z * 3 }}
        frameloop={still || !visible ? 'demand' : 'always'}
        style={{ width: '100%', height: '100%' }}
      >
        <Scene input={input} still={still} supersample={supersample} onReady={onReady} />
      </Canvas>
    </div>
  )
}
