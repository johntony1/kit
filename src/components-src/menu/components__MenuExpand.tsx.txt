/* ─────────────────────────────────────────────────────────
 * MENU EXPAND — animation storyboard
 *
 *  CARD            46 (pill) → 183 (open)
 *                        + 152 per expanded section (335, 487)
 *  EASE            cubic-bezier(0.22, 1, 0.36, 1) — single curve
 *
 *  OPEN
 *      0ms        pill content blurs+fades out (100ms)
 *      0ms        card resize tween starts
 *    140ms        dashboard row enters (y/blur/opacity)
 *   +60ms         profile, resources, settings rows stagger in
 *
 *  EXPAND ANY SECTION (Dashboard or Profile)
 *      0ms        chevron rotates 180° on the resize ease
 *      0ms        drawer height 0 → 142, margin-top 0 → 10
 *    120ms        first sub-row enters
 *   +50ms         remaining sub-rows stagger
 *
 *  CLOSE
 *    Rows fade out together (no stagger), drawer collapses,
 *    card shrinks back — all on the same easing curve.
 * ───────────────────────────────────────────────────────── */

function Icon({ box }: { box: IconBox }) {
  return (
    <span className="menu-icon" aria-hidden="true">
      <span style={{ inset: box.inset }}>
        <img src={box.src} alt="" />
      </span>
    </span>
  )
}

function Chevron() {
  return (
    <span className="menu-chevron" aria-hidden="true">
      <span className="menu-icon">
        <span style={{ inset: '35.83% 26.13% 35% 26.14%' }}>
          <img src={chevronIcon} alt="" />
        </span>
      </span>
    </span>
  )
}

export function MenuExpand() {
  const [open, setOpen] = useState(false)
  const [openSections, setOpenSections] = useState<Set<SectionId>>(new Set())

  const isSectionOpen = (id: SectionId) => openSections.has(id)

  const toggleSection = (id: SectionId, e: React.MouseEvent) => {
    e.stopPropagation()
    setOpenSections((prev) => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return next
    })
  }

  const handleCardClick = () => {
    if (!open) setOpen(true)
  }

  const cardClasses = ['menu-card', open ? 'is-open' : '']
    .filter(Boolean)
    .join(' ')

  const cardStyle = {
    ['--open-count' as string]: openSections.size,
  } as React.CSSProperties

  const closeAll = (e: React.MouseEvent) => {
    e.stopPropagation()
    setOpenSections(new Set())
    setOpen(false)
  }

  return (
    <div className={`menu-wrapper ${open ? 'is-open' : ''}`}>
    <div
      className={cardClasses}
      style={cardStyle}
      onClick={handleCardClick}
      role={open ? undefined : 'button'}
      tabIndex={open ? -1 : 0}
      onKeyDown={(e) => {
        if (!open && (e.key === 'Enter' || e.key === ' ')) {
          e.preventDefault()
          setOpen(true)
        }
        if (open && e.key === 'Escape') {
          setOpenSections(new Set())
          setOpen(false)
        }
      }}
      aria-expanded={open}
      aria-label={open ? 'Menu' : 'Open menu'}
    >
      <div className="menu-card__layer menu-card__layer--collapsed">
        <Icon box={{ src: menuIcon, inset: '20% 16.25% 20% 16.25%' }} />
        <span className="menu-collapsed__label">Menu</span>
      </div>

      <ul className="menu-card__layer menu-card__layer--expanded">
        {MAIN_ITEMS.map((item) => {
          const expandable = !!item.subItems
          const sectionId = item.id as SectionId
          const sectionState =
            expandable && isSectionOpen(sectionId) ? 'open' : 'closed'

          const row = (
            <div
              className={[
                'menu-row',
                item.elevated ? 'menu-row--elevated' : '',
                expandable ? 'menu-row--expandable' : '',
              ]
                .filter(Boolean)
                .join(' ')}
              data-row={item.id}
              style={{ borderRadius: item.radius }}
              onClick={
                expandable ? (e) => toggleSection(sectionId, e) : undefined
              }
            >
              <span className="menu-row__group">
                <Icon box={item.icon} />
                <span className="menu-row__label">{item.label}</span>
              </span>
              <Chevron />
            </div>
          )

          if (expandable && item.subItems) {
            return (
              <li
                key={item.id}
                className="menu-section"
                data-section={item.id}
                data-state={sectionState}
                style={{ listStyle: 'none' }}
              >
                {row}
                <div
                  className="menu-submenu"
                  role="region"
                  aria-label={`${item.label} sub-menu`}
                >
                  {item.subItems.map((sub) => (
                    <div key={sub.id} className="submenu-row">
                      <Icon box={sub.icon} />
                      <span className="submenu-row__label">{sub.label}</span>
                    </div>
                  ))}
                </div>
              </li>
            )
          }

          return (
            <li key={item.id} style={{ listStyle: 'none', width: '100%' }}>
              {row}
            </li>
          )
        })}
      </ul>
    </div>
      <button
        type="button"
        className="menu-close"
        onClick={closeAll}
        aria-label="Close menu"
        tabIndex={open ? 0 : -1}
      >
        <svg viewBox="0 0 24 24" fill="none" aria-hidden="true">
          <path
            d="M6 6L18 18M18 6L6 18"
            stroke="currentColor"
            strokeWidth="2.25"
            strokeLinecap="round"
          />
        </svg>
      </button>
    </div>
  )
}
