import { useMemo, useRef, useState, type KeyboardEvent } from 'react'
import { AnimatePresence, motion } from 'motion/react'
import { Icon } from '@iconify/react'

type User = {
  name: string
  email: string
  avatar?: string
}

const ALL_USERS: User[] = [
  { name: 'John Smith', email: 'john.smith@email.com' },
  { name: 'Jane Doe', email: 'jane.doe@email.com' },
  { name: 'Joel Castro', email: 'joel.castro@email.com' },
  { name: 'Jenny Park', email: 'jenny.park@email.com' },
  { name: 'Jordan Lee', email: 'jordan.lee@email.com' },
  { name: 'Julia Chen', email: 'julia.chen@email.com' },
  { name: 'Nitish Khagwal', email: 'nitish.khagwal@email.com' },
]

const isEmailish = (s: string) => /\S+@\S+\.\S+/.test(s.trim())

/* ─────────────────────────────────────────────────────────
 * SHARE CARD — chip input, autocomplete, two button states
 *
 *   Empty (no pills, no query) ─→ blue Paste button
 *   Has query OR pills         ─→ black circular send arrow
 *
 *   • placeholder "Email Address"
 *   • typing matches against ALL_USERS, dropdown springs in
 *   • Enter / click → email becomes a pill chip in the row
 *   • Backspace on empty → removes last pill
 *   • Esc clears query
 * ───────────────────────────────────────────────────────── */
export function ShareCard({ onClose }: { onClose: () => void }) {
  const [query, setQuery] = useState('')
  const [pills, setPills] = useState<User[]>([])
  const [highlighted, setHighlighted] = useState(0)
  const inputRef = useRef<HTMLInputElement>(null)

  const suggestions = useMemo(() => {
    const q = query.trim().toLowerCase()
    if (!q) return [] as User[]
    return ALL_USERS.filter(
      (u) =>
        !pills.some((p) => p.email === u.email) &&
        (u.name.toLowerCase().includes(q) ||
          u.email.toLowerCase().includes(q)),
    ).slice(0, 5)
  }, [query, pills])

  const dropdownOpen = suggestions.length > 0
  const isEmpty = pills.length === 0 && query.length === 0

  const addPill = (user: User) => {
    setPills((prev) =>
      prev.some((p) => p.email === user.email) ? prev : [...prev, user],
    )
    setQuery('')
    setHighlighted(0)
    inputRef.current?.focus()
  }

  const removePill = (email: string) => {
    setPills((prev) => prev.filter((p) => p.email !== email))
    inputRef.current?.focus()
  }

  const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'Backspace' && query === '' && pills.length > 0) {
      e.preventDefault()
      setPills(pills.slice(0, -1))
    } else if (e.key === 'ArrowDown' && dropdownOpen) {
      e.preventDefault()
      setHighlighted((i) => Math.min(i + 1, suggestions.length - 1))
    } else if (e.key === 'ArrowUp' && dropdownOpen) {
      e.preventDefault()
      setHighlighted((i) => Math.max(i - 1, 0))
    } else if (e.key === 'Enter') {
      e.preventDefault()
      if (dropdownOpen && suggestions[highlighted]) {
        addPill(suggestions[highlighted])
      } else if (isEmailish(query)) {
        addPill({
          name: query.trim().split('@')[0],
          email: query.trim(),
        })
      }
    } else if (e.key === 'Escape') {
      setQuery('')
      setHighlighted(0)
    }
  }

  const handlePaste = async () => {
    try {
      const text = (await navigator.clipboard.readText()).trim()
      if (!text) {
        inputRef.current?.focus()
        return
      }
      if (isEmailish(text)) {
        addPill({ name: text.split('@')[0], email: text })
      } else {
        setQuery(text)
        inputRef.current?.focus()
      }
    } catch {
      inputRef.current?.focus()
    }
  }

  const handleSend = () => {
    if (isEmailish(query)) {
      addPill({ name: query.trim().split('@')[0], email: query.trim() })
    }
    if (pills.length > 0) {
      setPills([])
      setQuery('')
    }
  }

  return (
    <div className="flex w-full flex-col gap-4">
      <div className="flex items-center justify-between">
        <h2
          id="share-title"
          className="text-text-strong-950"
          style={{
            fontWeight: 475,
            fontSize: 16,
            lineHeight: '22px',
            letterSpacing: '-0.18px',
          }}
        >
          Add people
        </h2>
        <button
          type="button"
          onClick={onClose}
          aria-label="Close share"
          className="-mr-1 grid size-7 place-items-center rounded-md text-text-sub-600 outline-none transition-[scale,color] hover:text-text-strong-950 active:scale-[0.92] focus-visible:ring-2 focus-visible:ring-black/15"
          style={{ willChange: 'transform' }}
        >
          <Icon icon="mingcute:close-line" width={16} height={16} />
        </button>
      </div>

      <div className="relative w-full">
        <div
          onClick={() => inputRef.current?.focus()}
          className="relative flex min-h-[48px] cursor-text items-center gap-2 rounded-full bg-white py-1.5 pl-3 pr-1.5"
          style={{
            boxShadow:
              '0 2px 4px 0 rgba(0,0,0,0.04), 0 1px 2px 0 rgba(0,0,0,0.06), 0 0 1px 0.5px var(--color-stroke-soft-200)',
          }}
        >
          <div className="flex min-w-0 flex-1 flex-wrap items-center gap-1.5">
            {pills.map((user) => (
              <PillChip
                key={user.email}
                user={user}
                onRemove={() => removePill(user.email)}
              />
            ))}
            <input
              ref={inputRef}
              type="text"
              value={query}
              onChange={(e) => {
                setQuery(e.target.value)
                setHighlighted(0)
              }}
              onKeyDown={handleKeyDown}
              placeholder={pills.length === 0 ? 'Email Address' : ''}
              aria-label="Email to invite"
              autoComplete="off"
              className="min-w-[100px] flex-1 bg-transparent px-1 py-1 text-[15px] leading-[20px] font-normal text-text-strong-950 placeholder:text-text-sub-600 outline-none"
            />
          </div>

          <AnimatePresence mode="popLayout" initial={false}>
            {isEmpty ? (
              <motion.div
                key="paste"
                initial={{ opacity: 0, scale: 0.6 }}
                animate={{ opacity: 1, scale: 1 }}
                exit={{ opacity: 0, scale: 0.6 }}
                transition={{
                  type: 'spring',
                  visualDuration: 0.24,
                  bounce: 0.18,
                }}
                style={{ willChange: 'transform' }}
              >
                <PasteButton onClick={handlePaste} />
              </motion.div>
            ) : (
              <motion.div
                key="send"
                initial={{ opacity: 0, scale: 0.6 }}
                animate={{ opacity: 1, scale: 1 }}
                exit={{ opacity: 0, scale: 0.6 }}
                transition={{
                  type: 'spring',
                  visualDuration: 0.24,
                  bounce: 0.18,
                }}
                style={{ willChange: 'transform' }}
              >
                <SendButton onClick={handleSend} />
              </motion.div>
            )}
          </AnimatePresence>
        </div>

        <AnimatePresence>
          {dropdownOpen && (
            <Dropdown
              suggestions={suggestions}
              highlighted={highlighted}
              onSelect={addPill}
              onHover={setHighlighted}
            />
          )}
        </AnimatePresence>
      </div>
    </div>
  )
}

function PillChip({
  user,
  onRemove,
}: {
  user: User
  onRemove: () => void
}) {
  return (
    <motion.span
      initial={{ opacity: 0, scale: 0.6 }}
      animate={{ opacity: 1, scale: 1 }}
      transition={{
        type: 'spring',
        visualDuration: 0.22,
        bounce: 0.15,
      }}
      className="inline-flex h-[34px] shrink-0 items-center gap-2 rounded-full bg-white pl-1 pr-1.5"
      style={{
        boxShadow:
          '0 0 0 1px rgba(51,51,51,0.04), 0 16px 8px -8px rgba(51,51,51,0.01), 0 12px 6px -6px rgba(51,51,51,0.02), 0 5px 5px -2.5px rgba(51,51,51,0.08), 0 1px 3px -1.5px rgba(51,51,51,0.16), inset 0 -0.5px 0.5px 0 rgba(51,51,51,0.08)',
        willChange: 'transform',
      }}
    >
      <Avatar user={user} size={26} />
      <span
        className="text-text-strong-950"
        style={{
          fontFamily: 'Inter, sans-serif',
          fontSize: 14,
          lineHeight: '20px',
          fontWeight: 500,
          letterSpacing: '-0.06px',
        }}
      >
        {user.name}
      </span>
      <button
        type="button"
        onClick={(e) => {
          e.stopPropagation()
          onRemove()
        }}
        aria-label={`Remove ${user.name}`}
        className="grid size-[18px] shrink-0 place-items-center rounded-full text-icon-soft-400 outline-none transition-colors hover:text-text-sub-600 focus-visible:ring-2 focus-visible:ring-black/15"
      >
        <Icon icon="mingcute:close-circle-fill" width={16} height={16} />
      </button>
    </motion.span>
  )
}

function Dropdown({
  suggestions,
  highlighted,
  onSelect,
  onHover,
}: {
  suggestions: User[]
  highlighted: number
  onSelect: (u: User) => void
  onHover: (i: number) => void
}) {
  return (
    <motion.div
      role="listbox"
      initial={{ opacity: 0, y: -6, scale: 0.97, filter: 'blur(4px)' }}
      animate={{ opacity: 1, y: 0, scale: 1, filter: 'blur(0px)' }}
      exit={{ opacity: 0, y: -4, scale: 0.97, filter: 'blur(2px)' }}
      transition={{ type: 'spring', duration: 0.24, bounce: 0 }}
      style={{
        transformOrigin: 'top center',
        boxShadow:
          '0 0 0 1px rgba(0,0,0,0.04), 0 12px 32px -8px rgba(0,0,0,0.14), 0 4px 8px -4px rgba(0,0,0,0.08)',
        willChange: 'transform, filter',
      }}
      className="absolute left-0 right-0 top-[calc(100%+8px)] z-dropdown overflow-hidden rounded-2xl bg-white p-1.5"
    >
      {suggestions.map((u, i) => {
        const active = i === highlighted
        return (
          <button
            key={u.email}
            type="button"
            role="option"
            aria-selected={active}
            onPointerDown={(e) => {
              e.preventDefault()
              onSelect(u)
            }}
            onPointerEnter={() => onHover(i)}
            className="flex w-full items-center gap-2.5 rounded-xl px-2 py-2 text-left outline-none transition-colors"
            style={{
              backgroundColor: active
                ? 'var(--color-bg-weak-25)'
                : 'transparent',
            }}
          >
            <Avatar user={u} size={28} />
            <div className="flex min-w-0 flex-1 items-center justify-between gap-2">
              <span className="truncate text-[14px] font-medium leading-[20px] text-text-strong-950">
                {u.name}
              </span>
              <span className="shrink-0 text-[12px] leading-[18px] text-text-sub-600">
                {u.email}
              </span>
            </div>
          </button>
        )
      })}
    </motion.div>
  )
}

const GRADIENTS = [
  'linear-gradient(135deg, #FCD34D 0%, #34D399 100%)',
  'linear-gradient(135deg, #FB7185 0%, #FB923C 100%)',
  'linear-gradient(135deg, #60A5FA 0%, #A78BFA 100%)',
  'linear-gradient(135deg, #F472B6 0%, #818CF8 100%)',
  'linear-gradient(135deg, #2DD4BF 0%, #60A5FA 100%)',
  'linear-gradient(135deg, #FBBF24 0%, #F87171 100%)',
] as const

const pickGradient = (seed: string): string => {
  let h = 0
  for (let i = 0; i < seed.length; i++) {
    h = (h * 31 + seed.charCodeAt(i)) >>> 0
  }
  return GRADIENTS[h % GRADIENTS.length]
}

function Avatar({ user, size = 24 }: { user: User; size?: number }) {
  if (user.avatar) {
    return (
      <img
        src={user.avatar}
        alt=""
        width={size}
        height={size}
        className="shrink-0 rounded-full bg-bg-soft-200 object-cover"
        style={{
          width: size,
          height: size,
          boxShadow: 'inset 0 0 0 1px rgba(0,0,0,0.1)',
        }}
      />
    )
  }
  return (
    <span
      aria-hidden
      className="grid shrink-0 place-items-center rounded-full font-semibold text-white"
      style={{
        width: size,
        height: size,
        fontSize: Math.max(10, Math.floor(size * 0.42)),
        background: pickGradient(user.email),
        boxShadow: 'inset 0 0 0 1px rgba(0,0,0,0.06)',
        textShadow: '0 1px 1px rgba(0,0,0,0.12)',
      }}
    >
      {user.name[0].toUpperCase()}
    </span>
  )
}
